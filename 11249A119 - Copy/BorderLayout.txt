
import javax.swing.*;
import java.awt.*;

public class BorderLayoutDemo {
    public static void main(String[] args) {
        JFrame frame = new JFrame("BorderLayout Demo");

        frame.setLayout(new BorderLayout(5, 5));

        frame.add(new JButton("Header - North"),
                  BorderLayout.NORTH);
        frame.add(new JButton("Footer - South"),
                  BorderLayout.SOUTH);
        frame.add(new JButton("Menu - West"),
                  BorderLayout.WEST);
        frame.add(new JButton("Options - East"),
                  BorderLayout.EAST);
        frame.add(new JButton("Main Content - Center"),
                  BorderLayout.CENTER);

        frame.setSize(500, 300);
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.setVisible(true);
    }
}
