interface Printable {
    void print();
}

interface Showable {
    void show();
}

class Demo implements Printable, Showable {
    public void print() {
        System.out.println("Printing information");
    }

    public void show() {
        System.out.println("Displaying information");
    }
}

public class MultipleInheritance {
    public static void main(String[] args) {
        Demo d = new Demo();

        d.print();
        d.show();
    }
}
